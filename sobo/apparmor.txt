#include <tunables/global>

# AppArmor profile of the SoBo app (plan 6). The Supervisor loads this file and renames
# the main profile to the full app slug (<repository>_sobo).
#
# The outer profile covers the s6 overlay and the bashio start script; the Python
# backend runs in the child profile `sobo_python` with only what it needs:
# network (Sonos, ingress, Supervisor API), read access to the runtime and
# read/write access to /data and /tmp.

profile sobo flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities. Note: `file,` also grants `ix` on every path, so the transition to
  # the backend profile below must use an exact path (exact rules take precedence).
  file,
  unix,
  signal (send) set=(kill,term,int,hup,cont),
  signal (receive) peer=unconfined,
  signal (receive) peer=sobo,
  signal (receive) peer=*_sobo,
  network inet stream,
  network inet6 stream,

  # S6 overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # App options and database
  /data/** rw,

  # The backend gets its own profile (python3 is a symlink to this binary; AppArmor
  # matches the resolved path). Keep in sync with the Python version of the base image.
  /usr/local/bin/python3.13 cx -> sobo_python,

  profile sobo_python flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # Receive signals from the s6 overlay
    signal (receive) peer=*_sobo,
    signal (receive) peer=sobo,
    signal (receive) peer=unconfined,

    # asyncio's self-pipe (socketpair); no bind/listen on unix sockets
    unix (create, getattr, getopt, setopt, read, write, send, receive, shutdown) type=stream,

    # Sonos control (TCP 1400), SSDP discovery (UDP multicast), ingress and the
    # internal API (listening sockets), Supervisor API; netlink for interface lookup
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network netlink raw,

    # Python runtime and installed packages (read-only)
    /usr/local/bin/python3* rix,
    /usr/local/lib/** mr,
    /usr/lib/** mr,
    /lib/** mr,
    deny /usr/local/lib/**/__pycache__/** w,

    # ctypes.util.find_library (used to enumerate network interfaces)
    /bin/busybox rix,
    /bin/sh rix,
    /sbin/ldconfig rix,

    # Configuration: TLS certificates, name resolution, time zones
    /etc/** r,
    /usr/share/zoneinfo/** r,
    /usr/share/ca-certificates/** r,

    # Kernel information (interfaces, CPU count)
    @{PROC}/** r,
    /sys/** r,

    /dev/null rw,
    /dev/tty rw,
    /dev/urandom r,
    /dev/random r,

    # App data (options.json, SQLite database incl. journal and locks, secret)
    /data/ r,
    /data/** rwk,
    /tmp/ r,
    /tmp/** rwk,
  }
}
